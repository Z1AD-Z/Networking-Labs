
! ==========================================
! Device: S2
! Role: Branch Office Access Switch
! Model: Cisco 2960-24TT
! ==========================================

hostname S2
no ip domain-lookup

!
! VLAN 1 Management Interface
!
interface vlan 1
 description BRANCH_OFFICE_SWITCH_MANAGEMENT
 ip address 192.168.30.2 255.255.255.0
 no shutdown
 exit

!
! Default Gateway
!
ip default-gateway 192.168.30.1

!
! Access Port to PC2
!
interface FastEthernet0/1
 description CONNECTION_TO_PC2
 switchport mode access
 switchport access vlan 1
 spanning-tree portfast
 no shutdown
 exit

!
! Access Port to R3
!
interface FastEthernet0/24
 description CONNECTION_TO_R3_GIGABITETHERNET0_0
 switchport mode access
 switchport access vlan 1
 no shutdown
 exit

!
! Console Configuration
!
line console 0
 logging synchronous
 exec-timeout 10 0
 exit

end
